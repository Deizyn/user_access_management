-- 2. เรียกดูพนักงานทั้งหมด (Clean Schema)

-- 3. เรียกดูข้อมูลกลุ่มสิทธิ์ทั้งหมด (ต้องใช้ Backtick ` ครอบคำว่า groups)

-- 4. เรียกดูพนักงานพร้อมสิทธิ์กลุ่มที่สังกัด (JOIN Query)
SELECT u.employee_id, u.display_name, u.company, g.group_name, g.category
FROM users u
JOIN user_groups ug ON u.employee_id = ug.employee_id
JOIN `groups` g ON ug.group_id = g.group_id;

-- 5. อัปเดตคอลัมน์ category ให้ตรงตามหมวดหมู่จริง

UPDATE `groups` SET `category` = 'NETWORK_VPN' WHERE `group_id` = 107;
